@startuml
skinparam ActivityShape rounded
skinparam ActivityBorderColor #333333
skinparam ActivityBackgroundColor #E0E0E0
skinparam PartitionBorderColor #333333
skinparam ArrowColor #333333

start

:Generate initial population using PairwiseTestCaseGenerator;

:Evaluate population via fitness function;

repeat
    :Evaluate current population;
    
    :Select elite population;
    
    :Form non-elite population as evaluated - elite;
    
    if (Onlooker selection enabled?) then (yes)
        :Apply probabilistic onlooker selection;
        :Update non-elite based on fitness scores;
    else (no)
        :Non-elite remains unchanged;
    endif
    
    :Apply mutations to non-elite population;
note right
  Temperature-controlled acceptance
  when EnforceMutationUniqueness=false
end note
    
    if (Scout phase enabled and generation > threshold?) then (yes)
        :Execute scout phase;
        :Replace poor performers with new solutions;
    else (no)
        :Continue with current population;
    endif
    
repeat while (More generations to process?)

:Apply final selection ratio to limit test suite size;

:Return optimized test suite;

stop
@enduml
